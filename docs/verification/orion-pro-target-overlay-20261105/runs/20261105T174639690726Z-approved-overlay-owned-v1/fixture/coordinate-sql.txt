                2,i,'absent','absent',%s,'stream:primary',true,%s,%s,%s,'provider',CASE WHEN i%%3=0 THEN 0.8200000000000001::float8 WHEN i%%3=1 THEN 0::float8 END, -25.01::float8, 'unknown'
